Функция ОтправитьДанныеПродажНаWebhook(ДанныеПродаж)
    
    // Преобразуем данные в JSON
    ЗаписьJSON = Новый ЗаписьJSON;
    ЗаписьJSON.УстановитьСтроку();
    ЗаписатьJSON(ЗаписьJSON, ДанныеПродаж);
    JSONСтрока = ЗаписьJSON.Закрыть();
    
    // Создаем подпись для безопасности
    Хеш = Новый ХешированиеДанных(ХешФункция.SHA256);
    Хеш.ДобавитьСтроку(JSONСтрока);
    ПодписьДанных = Хеш.ХешСумма;
    Подпись = "sha256=" + НРег(ПолучитьHexСтрокуИзДвоичныхДанных(ПодписьДанных));
    
    // Отправляем HTTP запрос
    HTTPСоединение = Новый HTTPСоединение(WebhookURL);
    HTTPЗапрос = Новый HTTPЗапрос("/webhook/sales");
    HTTPЗапрос.Заголовки.Вставить("Content-Type", "application/json; charset=utf-8");
    HTTPЗапрос.Заголовки.Вставить("X-Hub-Signature-256", Подпись);
    HTTPЗапрос.УстановитьТелоИзСтроки(JSONСтрока, КодировкаТекста.UTF8);
    
    Попытка
        HTTPОтвет = HTTPСоединение.ОтправитьДляОбработки(HTTPЗапрос);
        
        Если HTTPОтвет.КодСостояния = 200 Тогда
            Сообщить("✅ Данные продаж успешно отправлены");
            Возврат Истина;
        Иначе
            Сообщить("❌ Ошибка отправки: " + HTTPОтвет.КодСостояния);
            Возврат Ложь;
        КонецЕсли;
        
    Исключение
        Сообщить("❌ Ошибка соединения: " + ОписаниеОшибки());
        Возврат Ложь;
    КонецПопытки;
    
КонецФункции
